import numpy as np
import matplotlib.pyplot as plt

n_max = 500
np.random.seed(42)

# Вероятность 0.7
throws = np.random.binomial(1, 0.7, n_max)
running_avg = np.cumsum(throws) / np.arange(1, n_max + 1)

plt.figure(figsize=(10, 4.5))
plt.plot(running_avg, color='steelblue', linewidth=1.5, alpha=0.8)
plt.axhline(0.7, color='red', linestyle='--', linewidth=2.5, label='p = 0,7')
plt.xlabel('Число бросков')
plt.ylabel('Доля орлов')
plt.legend()
plt.grid(alpha=0.3)
plt.ylim(0.5, 0.9)
plt.show()

4. Эмпирическая функция распределения

Показать код 
import numpy as np
import matplotlib.pyplot as plt
import pandas as pd
from ipywidgets import Button, Output, VBox

# Исходная выборка
X = [0, 11, 2, 3, 9, 2, 8, 6, 3, 4, 8, 7, 5, 9, 4, 8, 6]

# Виджеты
out = Output()
btn1 = Button(description="1. Выборка", button_style='info')
btn2 = Button(description="2. Вариационный ряд", button_style='success')
btn3 = Button(description="3. Таблица вероятностей", button_style='warning')
btn4 = Button(description="4. Эмпирическая функция распределения", button_style='danger')

# Шаги (по кнопкам)
@btn1.on_click
def show_sample(b):
    with out:
        print("1. Исходная выборка:")
        print("   ", X, "\n")

@btn2.on_click
def show_sorted(b):
    with out:
        sorted_X = sorted(X)
        print("2. Вариационный ряд:")
        print("   ", sorted_X, "\n")

@btn3.on_click
def show_probs(b):
    with out:
        n = len(X)
        unique_vals, counts = np.unique(X, return_counts=True)
        probs = np.round(counts / n, 2)
        df = pd.DataFrame({'xᵢ': unique_vals, 'nᵢ': counts, 'pᵢ': probs})
        display(df.style.format({'pᵢ': '{:.2f}'}).set_properties(**{'text-align': 'center'}).hide(axis='index'))
        print()

@btn4.on_click
def show_ecdf(b):
    with out:
        n = len(X)
        unique_vals, counts = np.unique(X, return_counts=True)
        probs = counts / n
        cum_probs = np.cumsum(probs)

        x_step = np.concatenate(([min(X)-1], unique_vals))
        y_step = np.concatenate(([0], cum_probs))

        plt.figure(figsize=(8.5, 3.8))

        # Горизонтальные участки
        for i in range(len(x_step)-1):
            plt.plot([x_step[i], x_step[i+1]], [y_step[i], y_step[i]],
                     color='blue', linewidth=2)

        # Вертикальные скачки (пунктир)
        for i in range(1, len(x_step)):
            plt.plot([x_step[i], x_step[i]], [y_step[i-1], y_step[i]],
                     color='blue', linestyle='--', linewidth=1.5, alpha=0.7)

        # Точки: закрашенные (справа) и выколотые (слева)
        prev = np.concatenate(([0], cum_probs[:-1]))
        plt.scatter(unique_vals, cum_probs, color='blue', s=45, zorder=5, marker='s')
        plt.scatter(unique_vals, prev, facecolors='none', edgecolors='blue', s=45, zorder=6, marker='s')

        plt.title('Эмпирическая функция распределения', fontsize=12, fontweight='bold')
        plt.xlabel('x')
        plt.ylabel('F(x)')
        plt.grid(True, linestyle='--', alpha=0.7)
        plt.ylim(-0.05, 1.05)
        plt.xlim(min(X)-1.5, max(X)+1.5)
        plt.tight_layout()
        plt.show()

6. Гистограмма (6 интервалов) + полигон

Показать код 
import numpy as np
import matplotlib.pyplot as plt
import pandas as pd
from ipywidgets import Button, Output, VBox

X = [0, 11, 2, 3, 9, 2, 8, 6, 3, 4, 8, 7, 5, 9, 4, 8, 6]

out = Output()
btn1 = Button(description="1. Выборка", button_style='info')
btn2 = Button(description="2. Вариационный ряд", button_style='success')
btn3 = Button(description="3. Таблица (4 интервала)", button_style='primary')
btn4 = Button(description="4. Гистограмма (4 интервала)", button_style='primary')
btn5 = Button(description="5. Таблица (6 интервалов)", button_style='warning')
btn6 = Button(description="6. Гистограмма (6 интервалов) + полигон", button_style='warning')

def fmt_interval(a, b):
    # формат: [0,0; 2,8)
    return f"[{a:.1f}; {b:.1f})".replace('.', ',')

@btn3.on_click
def group_4(b):
    with out:
        bins_4 = np.linspace(min(X), max(X), 5)
        counts_4, _ = np.histogram(X, bins=bins_4)
        probs_4 = np.round(counts_4 / len(X), 2)
        intervals_4 = [fmt_interval(bins_4[i], bins_4[i+1]) for i in range(4)]
        df_4 = pd.DataFrame({'Интервал': intervals_4, 'nᵢ': counts_4, 'pᵢ': probs_4})
        display(df_4.style.format({'pᵢ': '{:.2f}'}).set_properties(**{'text-align': 'center'}).hide(axis='index'))

@btn4.on_click
def hist_4(b):
    with out:
        bins_4 = np.linspace(min(X), max(X), 5)
        width_4 = bins_4[1] - bins_4[0]
        counts_4, _ = np.histogram(X, bins=bins_4)
        probs_4 = counts_4 / len(X)

        plt.figure(figsize=(8, 3.6))
        heights, _, _ = plt.hist(X, bins=bins_4, edgecolor='black', alpha=0.8, density=True)

        # подписи вероятностей над столбцами
        for i, prob in enumerate(probs_4):
            plt.text(bins_4[i] + width_4/2, heights[i] + 0.05*max(heights), f"{prob:.2f}".replace('.', ',').replace('.', ','),
                     ha='center', fontsize=9, weight='bold')

        plt.title('Гистограмма (4 интервала)', fontsize=11)
        plt.xlabel('Значения')
        plt.ylabel('Высота')
        plt.grid(True, linestyle='--', alpha=0.5)
        plt.ylim(0, 1.35*max(heights))
        plt.tight_layout()
        plt.show()

@btn6.on_click
def hist_6(b):
    with out:
        bins_6 = np.linspace(min(X), max(X), 7)
        width_6 = bins_6[1] - bins_6[0]
        counts_6, _ = np.histogram(X, bins=bins_6)
        probs_6 = counts_6 / len(X)

        plt.figure(figsize=(8, 3.6))
        heights, _, _ = plt.hist(X, bins=bins_6, edgecolor='black', alpha=0.8, density=True)

        for i, prob in enumerate(probs_6):
            plt.text(bins_6[i] + width_6/2, heights[i] + 0.05*max(heights), f"{prob:.2f}".replace('.', ',').replace('.', ','),
                     ha='center', fontsize=9, weight='bold')

        # полигон
        midpoints_6 = (bins_6[:-1] + bins_6[1:]) / 2
        poly_heights = probs_6 / width_6
        plt.plot(midpoints_6, poly_heights, 'o-', color='red', linewidth=2, markersize=6, label='Полигон')
        plt.legend()

        plt.title('Гистограмма (6 интервалов) + полигон', fontsize=11)
        plt.xlabel('Значения')
        plt.ylabel('Высота')
        plt.grid(True, linestyle='--', alpha=0.5)
        plt.ylim(0, 1.35*max(heights))
        plt.tight_layout()
        plt.show()

display(VBox([btn1, btn2, btn3, btn4, btn5, btn6]))
display(out)
